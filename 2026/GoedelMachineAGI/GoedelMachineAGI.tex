  \begin{block}{\textbf{Theorem 4.1}} \emph{[...] assuming
      {\color{red}consistency} of the underlying formal system [...]
      the utility of starting the execution of the present
      {\color{red}[rewrite] is higher than the utility of waiting for
        the proof searcher to produce an alternative [rewrite]}

  \end{block}

  \vspace{4mm}

\begin{frame}[fragile]
  \frametitle{What is in the engine?}
  \begin{itemize}
  \item Get axioms from the theory:
  \begin{minted}[fontsize=\scriptsize]{racket}
(get_axiom 42)    ; Retrieve 42th axiom (first Kolmogorov axiom)
  \end{minted}
$\leadsto$
  \begin{minted}[fontsize=\scriptsize]{racket}
(: K1                       ; Symbol of first Kolmogorov axiom
   (-> (: $o Set)           ; Premise 1: $o is a sample space
       (: $f (Field $o))    ; Premise 2: $f is a field
       (: $p (Measure $f))  ; Premise 3: $p is a probability measure
       (: $e $f)            ; Premise 4: $e is an event
       (<= 0 ($p $e))))     ; Conclusion: ($p $e) is non-negative
  \end{minted}
  \item Get proprioceptive axioms:
  \begin{minted}[fontsize=\scriptsize]{racket}
(get_proprio_axiom_at_cell 4095)  ; Retrieve memory content of
                                  ; cell 4095 and deliver it as axiom
  \end{minted}
$\leadsto$
  \begin{minted}[fontsize=\scriptsize]{racket}
(: MC4095@2026-11-05:06:58:12:345
   (= (MemoryContent 4095 2026-11-05:06:58:12:345) 256))
  \end{minted}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]
  \frametitle{What is in the engine?}
  \begin{itemize}
  \item Chainer:
    \begin{minted}[fontsize=\scriptsize]{racket}
;; Call chainer on target theorem
(chain PARAMETERS (: $proof (>= (E (u (rewrite $r $t))) (E (u no_rewrite)))))
  \end{minted}
  $\leadsto$
