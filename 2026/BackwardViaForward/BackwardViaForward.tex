\begin{frame}
  MORK (MeTTa Optimal Reduction Kernel):
  \begin{itemize}
  \item MeTTa: {\color{red}functional logic} programming language
  \item Minimal MeTTa 2 (MM2): query language with
    {\color{red}unification and rewrite} commands
  \item Strengths of MORK:
    \begin{itemize}
    \item {\color{red}compress data into a trie}
    \item absurd {\color{red}speed-ups} on some queries
    \item {\color{red}Faster} than PeTTa (Prolog based back-end for
      MeTTa) on {\color{red}forward chaining}.
    \end{itemize}
  \item Weaknesses of MORK:
    \begin{itemize}
    \item No stack (everything happens on the trie)
    \item Magnitudes of times {\color{red}slower} than PeTTa on
      {\color{red}backward chaining}.
    \end{itemize}
  \end{itemize}
\end{frame}

  \item Backward chaining: theorems $\leadsto$ axioms
  \end{itemize}

\begin{frame}
  \frametitle{Forward vs Backward Chaining}

  \begin{center}
    Can you emulate backward chaining with forward chaining?

    \pause

    \vspace{1cm}

    {\color{red}Yes!}

    \vspace{0.5cm}

    Using the {\color{red}blackbird} combinator
    $$\blackbird : (\cv \arrowtype \dv) \arrowtype (\av \arrowtype \bv \arrowtype \cv) \arrowtype \av \arrowtype \bv \arrowtype \dv$$
  \end{center}
\end{frame}

\begin{frame}
  \frametitle{Emulating Backward Chaining with Forward Chaining}

  \begin{itemize}
  \item Let $\Gamma$ be a theory
  \item Let $\Tau$ be a target theorem of $\Gamma$
  \item Find $(\Gamma^i, \Tau^i)$ such that forward chaining on
    ($\Gamma^i, \Tau^i)$ behaves like backward chaining on
    ($\Gamma$, $\Tau$)
  \end{itemize}

  \pause

  \begin{eqnarray*}
    \Gamma^i &=& \textit{Axioms}(\Gamma) \cup
                    \textit{Rules}(\Gamma)^i \\
    \Tau^i &=& \idiot : \Tau \arrowtype \Tau
  \end{eqnarray*}
\end{frame}

\begin{frame}
  \frametitle{Emulating Backward Chaining with Forward Chaining}

  \begin{eqnarray*}
    \blackbird &:& (\cv \arrowtype \dv) \arrowtype (\av \arrowtype \bv
                   \arrowtype \cv) \arrowtype \av \arrowtype \bv
                   \arrowtype \dv
  \end{eqnarray*}

  \begin{columns}
    \column{4cm}
    \includegraphics[scale=0.2]{figs/proof-tree-bef-exp.png}
    \column{0.5cm}
    $\arrowtype$
    \column{2cm}
    \begin{prooftree}
      \AxiomC{$\av$}
      \AxiomC{$\bv$}
      \RightLabel{(R)}
      \BinaryInfC{$\cv$}
    \end{prooftree}
    \column{0.5cm}
    $\arrowtype$
    \column{4cm}
    \includegraphics[scale=0.2]{figs/proof-tree-aft-exp.png}
  \end{columns}

  \pause

  % \emph{\underline{Invert rule:}}
  \begin{eqnarray*}
    R^i &=& \lambda x.(\blackbird\ x\ R) \\
    % R &:& \av \arrowtype \bv \arrowtype \cv \\
    R^i &:& (\cv \arrowtype \dv) \arrowtype \av \arrowtype \bv \arrowtype \dv
  \end{eqnarray*}
\end{frame}

\begin{frame}
  \frametitle{Emulating Backward Chaining with Forward Chaining:
    Example}
  {\small
  \underline{Fragment of propositional calculus for backward chaining:}
  \begin{eqnarray*}
    \Gamma = &\{& \axon : \phi \to (\psi \to \phi), \\
             & & \axtw : ( \phi \to ( \psi \to \chi ) ) \to ( ( \phi
                 \to \psi ) \to ( \phi \to \chi ) ), \\
             & & \axmp : ( \phi \to \psi ) \arrowtype \phi \arrowtype
                 \psi \\
             &\}& \\
    % \Tau = & & ? : (\av \to (\av \to \bv)) \to (\av \to \bv)
  \end{eqnarray*}

  \underline{Inverted fragment of propositional calculus for forward chaining:}
  \begin{eqnarray*}
    \Gamma^i = &\{& \axon : \phi \to (\psi \to \phi), \\
               & & \axtw : ( \phi \to ( \psi \to \chi ) ) \to ( ( \phi
                   \to \psi ) \to ( \phi \to \chi ) ), \\
               & & \axmpi : (\bv \arrowtype \cv) \arrowtype (\av \to \bv) \arrowtype \av \arrowtype \cv \\
               &\}& \\
    % \Tau^i = & & \idiot : (\av \to (\av \to \bv)) \to (\av \to \bv) \arrowtype (\av \to (\av \to \bv)) \to (\av \to \bv)
  \end{eqnarray*}
  }
\end{frame}

\begin{frame}
  \frametitle{Emulating Backward Chaining with Forward Chaining:
    Example}

  Target theorem: $\av \to \av$

  \vspace{0.7cm}

  {\tiny
  \begin{prooftree}
    \AxiomC{$\ $}
    \RightLabel{$(\idiot)$}
    \UnaryInfC{$(\av \to \av) \arrowtype (\av \to \av)$}
    \RightLabel{{\color<1>{lightgray}$(\axmpi)$}}
    \UnaryInfC{{\color<1>{lightgray}$(\bv \to (\av \to \av)) \arrowtype \bv \arrowtype (\av \to \av)$}}
    \RightLabel{{\color<-2>{lightgray}$(\axmpi)$}}
    \AxiomC{$\ $}
    \RightLabel{{\color<-3>{lightgray}$(\axtw)$}}
    \UnaryInfC{{\color<-3>{lightgray}$( \av \to ( \bv \to \cv ) ) \to ( ( \av \to \bv ) \to ( \av \to \cv ) )$}}
    \AxiomC{$\ $}
    \RightLabel{{\color<-4>{lightgray}$(\axon)$}}
    \UnaryInfC{{\color<-4>{lightgray}$\av \to (\bv \to \av)$}}
    \AxiomC{$\ $}
    \RightLabel{{\color<-5>{lightgray}$(\axon)$}}
    \UnaryInfC{{\color<-5>{lightgray}$\av \to (\dv \to \av)$}}
    \RightLabel{{\color<-2>{lightgray}$(\axmpi)$}}
    \QuaternaryInfC{{\color<-6>{lightgray}$\av \to \av$}}
  \end{prooftree}
  }

  \vspace{0.7cm}

  \only<1>{$$\idiot : (\av \to \av) \arrowtype (\av \to \av)$$}
  \only<2>{$$(\axmpi\ \idiot) : (\bv \to (\av \to \av)) \arrowtype \bv \arrowtype (\av \to \av)$$}
  \only<3>{$$(\axmpi\ (\axmpi\ \idiot)) : (\cv \to (\bv \to (\av \to \av))) \arrowtype \cv \arrowtype \bv \arrowtype (\av \to \av)$$}
  \only<4>{$$((\axmpi\ (\axmpi\ \idiot))\ \axtw) : (\av \to (\bv \to \av)) \arrowtype (\av \to \bv) \arrowtype (\av \to \av)$$}
  \only<5>{$$(((\axmpi\ (\axmpi\ \idiot))\ \axtw)\ \axon) : (\av \to \bv) \arrowtype (\av \to \av)$$}
  \only<6-8>{$$((((\axmpi\ (\axmpi\ \idiot))\ \axtw)\ \axon)\ \axon) : \av \to \av$$}
  \visible<8>{
    % $$\downarrow$$
    $$\rotatebox{-90}{$\leadsto$}$$
    \vspace{0.04cm}
    $$(\axmp\ (\axmp\ \axtw\ \axon)\ \axon) : \av \to \av$$
  }
\end{frame}

\begin{frame}
  \frametitle{Emulating Backward Chaining with Forward Chaining:
    Example}

  \underline{Forward chaining:}\\
  {\scriptsize
  $(((\axmpi\ (((\axmpi\ (\axmpi\ \idiot))\ \axtw)\ \axtw))\ \axtw)\ \axon) : (\av \to (\av \to \bv)) \to (\av \to \bv)$\\
  $((((\axmpi\ (((\axmpi\ (\axmpi\ ((\axmpi\ ((\axmpi\ (\axmpi\ \idiot))\ \axtw))\ \axon)))\ \axtw)\ \axtw))\ \axon)\ \axon)\ \axon) : ((\av \to \bv) \to \cv) \to (\bv \to \cv)$\\
  $(((((\axmpi\ (\axmpi\ ((\axmpi\ ((\axmpi\ ((\axmpi\ (\axmpi\ \idiot))\ \axtw))\ \axon))\ \axtw)))\ \axtw)\ \axon)\ \axon)\ \axon) : \av \to ((\av \to \bv) \to \bv)$\\
  }

  \underline{Backward chaining:}\\
  {\scriptsize
  $(\axmp\ (\axmp\ \axtw\ \axtw)\ (\axmp\ \axtw\ \axon)) : (\av \to (\av \to \bv)) \to (\av \to \bv)$\\
  $(\axmp\ (\axmp\ \axtw\ (\axmp\ \axon\ (\axmp\ (\axmp\ \axtw\ \axtw)\ (\axmp\ \axon\ \axon))))\ \axon) : ((\av \to \bv) \to \cv) \to (\bv \to \cv)$\\
  $(\axmp\ (\axmp\ \axtw\ (\axmp\ \axon\ (\axmp\ \axtw\ (\axmp\ (\axmp\ \axtw\ \axon)\ \axon))))\ \axon) : \av \to ((\av \to \bv) \to \bv)$\\
  }
\end{frame}
