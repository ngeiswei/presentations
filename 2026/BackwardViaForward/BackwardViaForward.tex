    Can you emulate backward chaining with forward chaining?

    \pause

    \vspace{1cm}

    {\color{red}Yes!}

    \vspace{0.5cm}

    Using the {\color{red}blackbird} combinator
    $$\blackbird : (\cv \arrowtype \dv) \arrowtype (\av \arrowtype \bv \arrowtype \cv) \arrowtype \av \arrowtype \bv \arrowtype \dv$$
  \end{center}

\begin{frame}
  \frametitle{Emulating Backward Chaining with Forward Chaining}

  \begin{itemize}
  \item Let $\Gamma$ be a theory
  \item Let $\Tau$ be a target theorem of $\Gamma$
  \item Find $(\Gamma^{-1}, \Tau^{-1})$ such that forward chaining on
    ($\Gamma^{-1}, \Tau^{-1})$ behaves like backward chaining on
    ($\Gamma$, $\Tau$)
  \end{itemize}

  \pause

  \begin{eqnarray*}
    \Gamma^{-1} &=& \textit{Axioms}(\Gamma) \cup
    \textit{Invert}(\textit{Rules}(\Gamma)) \\
    \Tau^{-1} &=& \idiot : \Tau \arrowtype \Tau
  \end{eqnarray*}
\end{frame}
